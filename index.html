        TRUY VẤN 1 · MILESTONE QUÁ HẠN

        1. Lọc Milestone có due_date < CURRENT_DATE và status <> 'APPROVED'

        SELECT m.id, c.title AS contract_title, m.title, m.amount,
       m.due_date, m.status, (CURRENT_DATE - m.due_date) AS days_overdue
FROM miniupwork.milestones m
JOIN miniupwork.contracts c ON c.id = m.contract_id
WHERE m.due_date < CURRENT_DATE AND m.status <> 'APPROVED';
        IDCột mốcHạn chótTrạng tháiTrễ hạn
        TRUY VẤN 2 · GIỜ CHƯA LẬP HÓA ĐƠN

        2. Lọc Time Entries có invoice_id IS NULL và status = 'APPROVED'

        SELECT t.id, c.title, t.work_date, t.hours, c.hourly_rate,
       ROUND(t.hours * c.hourly_rate, 2) AS unbilled_amount
FROM miniupwork.time_entries t
JOIN miniupwork.contracts c ON c.id = t.contract_id
WHERE t.invoice_id IS NULL AND t.status = 'APPROVED';
        IDNgàyMô tảGiờ × GiáTạm tính
        TRUY VẤN 3 · ĐỐI SOÁT THANH TOÁN

        3. LEFT JOIN Invoices sang Payments (SUCCESS), lọc hiệu số <> 0

        SELECT i.id, i.invoice_number, i.total_amount,
       COALESCE(SUM(p.paid_amount), 0) AS total_paid,
       (i.total_amount - COALESCE(SUM(p.paid_amount), 0)) AS discrepancy
FROM miniupwork.invoices i
LEFT JOIN miniupwork.payments p
  ON p.invoice_id = i.id AND p.status = 'SUCCESS'
GROUP BY i.id, i.invoice_number, i.total_amount
HAVING (i.total_amount - COALESCE(SUM(p.paid_amount), 0)) <> 0;
        Số HĐTổng HĐĐã thuLệchKết luận
        TRUY VẤN 4 · DOANH THU THEO KỲ

        4. Tổng hợp SUM(paid_amount) theo kỳ tháng/năm (status = 'SUCCESS')

        SELECT TO_CHAR(DATE_TRUNC('month', p.paid_at), 'YYYY-MM') AS period,
       COUNT(p.id) AS tx_count, SUM(p.paid_amount) AS total_revenue
FROM miniupwork.payments p
WHERE p.status = 'SUCCESS'
GROUP BY DATE_TRUNC('month', p.paid_at)
ORDER BY period;
        Kỳ (Tháng/Năm)Số giao dịch SUCCESSTổng doanh thu
